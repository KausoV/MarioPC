cmake_minimum_required(VERSION 3.24)
project(MarioPC LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(SDL3 CONFIG REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/SDL3-3.4.18/cmake" NO_DEFAULT_PATH)
set(CORE source/SMB-Vanilla/src/smbcore)
add_library(core_smb1 STATIC ${CORE}/common.c ${CORE}/common_sound.c ${CORE}/area.c ${CORE}/smb1only.c)
target_compile_definitions(core_smb1 PRIVATE SMB1_MODE PRINT_WARNINGS_AND_ERRORS)
add_library(core_smb2j STATIC ${CORE}/common.c ${CORE}/common_sound.c ${CORE}/area.c ${CORE}/smb2jonly.c)
target_compile_definitions(core_smb2j PRIVATE SMB2J_MODE PRINT_WARNINGS_AND_ERRORS)
add_library(mario_core STATIC ${CORE}/smbcore.c)
target_link_libraries(mario_core PUBLIC core_smb1 core_smb2j)
target_include_directories(mario_core PUBLIC source/SMB-Vanilla/src generated)
add_library(mario_raster STATIC source/SMB-Vanilla/src/platform/render_raster.c)
target_include_directories(mario_raster PUBLIC source/SMB-Vanilla/src)
add_library(mario_sound STATIC source/SMB-Vanilla/Nes_Snd_Emu/nes_apu/Blip_Buffer.cpp source/SMB-Vanilla/Nes_Snd_Emu/nes_apu/Nes_Apu.cpp source/SMB-Vanilla/Nes_Snd_Emu/nes_apu/Nes_Oscs.cpp source/SMB-Vanilla/src/platform/audio_sdl.cpp)
target_include_directories(mario_sound PUBLIC source/SMB-Vanilla/Nes_Snd_Emu source/SMB-Vanilla/src)
target_link_libraries(mario_sound PUBLIC SDL3::SDL3)
foreach(target core_smb1 core_smb2j mario_core mario_raster mario_sound)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /UndefIntOverflow-)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -fwrapv)
  endif()
endforeach()
set(IMGUI dependencies/imgui-1.92.9b)
add_library(mario_imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_sdl3.cpp ${IMGUI}/backends/imgui_impl_sdlrenderer3.cpp)
target_include_directories(mario_imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_link_libraries(mario_imgui PUBLIC SDL3::SDL3)
add_library(mario_settings STATIC src/pc_settings.cpp)
target_include_directories(mario_settings PUBLIC src source/SMB-Vanilla/src)
target_link_libraries(mario_settings PUBLIC SDL3::SDL3)
if(MSVC)
 target_compile_options(mario_imgui PRIVATE /utf-8)
 target_compile_options(mario_settings PRIVATE /utf-8)
 target_compile_definitions(mario_settings PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(mario_save STATIC src/game_save.cpp)
target_include_directories(mario_save PUBLIC src)
target_link_libraries(mario_save PUBLIC mario_core)
if(MSVC)
 target_compile_options(mario_save PRIVATE /utf-8)
endif()
add_library(mario_coin_art STATIC src/coin_art.cpp)
target_include_directories(mario_coin_art PUBLIC src)
if(MSVC)
 target_compile_options(mario_coin_art PRIVATE /utf-8)
endif()
add_library(mario_mushroom_art STATIC src/mushroom_art.cpp)

if(MSVC)

endif()
add_library(mario_character_art STATIC src/character_art.cpp src/chibi_art.cpp src/sheet_art.cpp src/world_background.cpp src/world_assets.cpp)
target_include_directories(mario_character_art PUBLIC src source/SMB-Vanilla/src)
target_link_libraries(mario_character_art PUBLIC mario_coin_art mario_mushroom_art mario_flower_art mario_camel_enemy_art mario_armadillo_enemy_art SDL3::SDL3)
if(MSVC)
 target_compile_options(mario_character_art PRIVATE /utf-8)
endif()
add_library(mario_frame_clock STATIC src/frame_clock.cpp)
target_include_directories(mario_frame_clock PUBLIC src)
if(MSVC)
 target_compile_options(mario_frame_clock PRIVATE /utf-8)
endif()
add_executable(MarioPC WIN32 src/mario_main.cpp src/settings_menu.cpp src/game_hud.cpp src/glass_panel.cpp)
target_link_libraries(MarioPC PRIVATE mario_core mario_raster mario_sound mario_settings mario_save mario_character_art mario_frame_clock mario_imgui SDL3::SDL3)
target_compile_definitions(MarioPC PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
 target_compile_options(MarioPC PRIVATE /utf-8)
endif()
add_custom_command(TARGET MarioPC POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:MarioPC>)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/native_checks.cpp")
 add_executable(MarioChecks tests/native_checks.cpp)
 target_link_libraries(MarioChecks PRIVATE mario_core mario_raster)
 if(MSVC)
  target_compile_options(MarioChecks PRIVATE /utf-8)
 endif()
 enable_testing()
 add_test(NAME NativeMario COMMAND MarioChecks "${CMAKE_CURRENT_BINARY_DIR}/verification")
endif()
